293def validate_story( 

294 packet: dict, 

295 value: dict, 

296) -> dict: 

297 """Validate direct citations, changed-region coverage, and prose bounds. 

298 

299 Args: 

300 packet: Validated source packet for the exact compared revisions. 

301 value: Untrusted parsed model output. 

302 

303 Returns: 

304 Detached, schema-validated story data. Structural acceptance does not 

305 establish that the prose is true. 

306 

307 Raises: 

308 ValueError: If schema, size, references, or coverage are invalid. 

309 """ 

310 if len(canonical_bytes(value)) > MAX_ANNOTATION_BYTES: 

311 msg = "Model annotation data exceeds the 1 MiB limit" 

312 raise ValueError(msg) 

313 try: 

314 story = Story.model_validate(value).model_dump() 

315 except ValueError as error: 

316 msg = "Invalid walkthrough story" 

317 raise ValueError(msg) from error 

318 validate_sections(packet, story["sections"]) 

319 if not story["introduction"].strip() or not story["closing"].strip(): 

320 msg = "Story requires nonblank introduction and closing" 

321 raise ValueError(msg) 

322 validate_editorial_fields(packet, story) 

323 return story

326def validate_editorial_fields(packet: dict, story: dict) -> None: 

327 """Check v3 opening, route, author, and review references against evidence. 

328 

329 Args: 

330 packet: Validated packet supplying prose, passages, and static tests. 

331 story: Story already accepted by the v3 structured schema. 

332 

333 Raises: 

334 ValueError: If text is blank, a reference is repeated or missing, an 

335 author excerpt is not exact, or a test is unrelated to its passage. 

336 

337 Essential behavior: 

338 Keeps generated summaries and review plans within the source boundary. 

339 """ 

340 passages = { 

341 passage["id"]: passage 

342 for section in story["sections"] 

343 for passage in section["passages"] 

344 } 

345 quick_summary = story["quick_summary"] 

346 if ( 

347 not quick_summary["before"].strip() 

348 or not quick_summary["after"].strip() 

349 ): 

350 msg = "Quick summary before and after statements must be nonblank" 

351 raise ValueError(msg) 

352 if story["impact"] is not None and not story["impact"]["text"].strip(): 

353 msg = "Impact text must be nonblank" 

354 raise ValueError(msg) 

355 references = [ 

356 ("quick summary", story["quick_summary"]["passage_ids"]), 

357 *[ 

358 (f"{route['kind']} route", route["passage_ids"]) 

359 for route in story["route"] 

360 ], 

361 ] 

362 if story["impact"] is not None: 

363 references.append(("impact", story["impact"]["passage_ids"])) 

364 for field, refs in references: 

365 _validate_passage_refs(field, refs, passages) 

366 _validate_routes_and_author(packet, story) 

367 _validate_review_checks(packet, story["review_checks"], passages)

370def _validate_routes_and_author(packet: dict, story: dict) -> None: 

371 """Require one mechanism route and exact author-description excerpts. 

372 

373 Args: 

374 packet: Validated packet with the author description. 

375 story: V3 story accepted by its strict structured schema. 

376 

377 Raises: 

378 ValueError: If route kinds repeat, mechanism is absent, or an author 

379 excerpt is not a nonblank exact substring of the description. 

380 

381 Essential behavior: 

382 It never creates or infers author-reported checks from other metadata. 

383 """

385 if ( 

386 len(set(route_kinds)) != len(route_kinds) 

387 or "mechanism" not in route_kinds 

388 ): 

389 msg = "Story routes must be unique and include a mechanism" 

390 raise ValueError(msg) 

391 description = packet["meta"].get("description") or "" 

392 for excerpt in story["author_reported_checks"]: 

393 if not excerpt.strip() or excerpt not in description: 

394 msg = "Author reported checks must quote the supplied description exactly" 

395 raise ValueError(msg)

398def _validate_review_checks( 

399 packet: dict, checks: list[dict], passages: dict[str, dict] 

400) -> None: 

401 """Require each remaining review question and optional test to be relevant. 

402 

403 Args: 

404 packet: Validated packet with static test source ranges. 

405 checks: V3 review checks accepted by the strict structured schema. 

406 passages: Final source-linked passage data indexed by passage ID. 

407 

408 Raises: 

409 ValueError: If prose or passage references are invalid, or a test is 

410 missing or does not overlap at least one referenced passage. 

411 

412 Essential behavior: 

413 Test relevance is checked against final, post-merge passage citations. 

414 """ 

415 tests = {test["id"]: test for test in packet["analysis"]["tests"]} 

416 for check in checks: 

417 if not check["text"].strip(): 

418 msg = "Review checks must be nonblank" 

419 raise ValueError(msg) 

420 _validate_passage_refs("review check", check["passage_ids"], passages) 

421 if len(set(check["test_ids"])) != len(check["test_ids"]): 

422 msg = "Review checks must not repeat static test IDs" 

423 raise ValueError(msg) 

424 for test_id in check["test_ids"]: 

425 test = tests.get(test_id) 

426 if test is None: 

427 msg = "Review check references an unknown static test" 

428 raise ValueError(msg) 

429 if not any( 

430 _citations_overlap(test["source"], citation) 

431 for passage_id in check["passage_ids"] 

432 for citation in passages[passage_id]["citations"] 

433 ): 

434 msg = "Review check references a static test unrelated to its passages" 

435 raise ValueError(msg)

438def _validate_passage_refs( 

439 label: str, refs: list[str], passages: dict[str, dict] 

440) -> None: 

441 """Require unique, nonblank passage IDs that resolve in the final story. 

442 

443 Args: 

444 label: User-visible category for the diagnostic. 

445 refs: Passage IDs selected by a story field. 

446 passages: Final source-linked passage data indexed by ID. 

447 

448 Raises: 

449 ValueError: If references are empty, repeated, blank, or unknown. 

450 

451 Essential behavior: 

452 All links resolve against the completed story after host remapping.

455 not refs

460 msg = (

466def _citations_overlap(left: dict, right: dict) -> bool: 

467 """Return whether two source ranges overlap in the same pinned buffer. 

468 

469 Args: 

470 left: First validated source range. 

471 right: Second validated source range. 

472 

473 Returns: 

474 True when both ranges name the same source/revision and share a line. 

475 

476 Essential behavior: 

477 A matching path or line number alone is not sufficient. 

478 """ 

479 return ( 

480 left["source_id"] == right["source_id"] 

481 and left["revision"] == right["revision"] 

482 and left["start"] <= right["end"] 

483 and right["start"] <= left["end"] 

484 )

487def validate_sections(packet: dict, sections: list[dict]) -> None: # noqa: C901 # Validate citation support and full coverage in one bounded pass. 

488 """Require schema-validated sections to account for all assigned evidence. 

489 

490 Args: 

491 packet: Validated complete evidence or a writer's assigned subset. 

492 sections: Sections already validated by Story or WritingPart. No 

493 synthetic document framing is needed for a writing part. 

494 

495 Raises: 

496 ValueError: If citations, unique IDs, or change/region coverage fail. 

497 """ 

498 sources = source_index(packet) 

499 changes = {c["id"] for c in packet["changes"]} 

500 regions = {r["id"]: r for r in packet["regions"]} 

501 seen_sections, seen_passages, seen_changes, seen_regions = ( 

502 set(), 

503 set(), 

504 set(), 

505 set(), 

506 ) 

507 for section in sections: 

508 if section["id"] in seen_sections or not section["title"].strip(): 

509 msg = "Story section IDs must be unique and titles nonblank" 

510 raise ValueError(msg) 

511 seen_sections.add(section["id"]) 

512 for passage in section["passages"]: 

513 if passage["id"] in seen_passages or not passage["text"].strip(): 

514 msg = "Story passage IDs must be unique and prose nonblank" 

515 raise ValueError(msg) 

516 seen_passages.add(passage["id"]) 

517 for field, allowed in ( 

518 ("change_ids", changes), 

519 ("region_ids", set(regions)), 

520 ): 

521 refs = passage[field] 

522 if len(set(refs)) != len(refs) or not set(refs) <= allowed: 

523 msg = "Story accounts for duplicate or unknown changes/regions" 

524 raise ValueError(msg) 

525 for citation in passage["citations"]: 

526 resolve_citation(citation, sources) 

527 for region_id in passage["region_ids"]: 

528 region = regions[region_id] 

529 if not any( 

530 a["source_id"] == b["source_id"] 

531 and a["start"] <= b["end"] 

532 and b["start"] <= a["end"] 

533 for a in passage["citations"] 

534 for b in region["citations"] 

535 ): 

536 msg = ( 

537 "An accounted region has no supporting source citation" 

538 ) 

539 raise ValueError(msg) 

540 seen_changes.update(passage["change_ids"]) 

541 seen_regions.update(passage["region_ids"]) 

542 if seen_changes != changes or seen_regions != set(regions): 

543 msg = "Story does not account for every supplied change and changed region" 

544 raise ValueError(msg)

547def validate_walkthrough(report: dict) -> None: 

548 """Validate a flat saved report using source and story boundaries only. 

549 

550 Args: 

551 report: Untrusted decoded diffstory.report.v2 artifact. 

552 

553 Returns: 

554 ``None`` when the complete flat report passes validation. 

555 

556 Raises: 

557 ValueError: If schema, primitive fields, revisions, coverage, 

558 citations, usage, or state fail. 

559 

560 Preconditions: 

561 ``report`` is parsed data and may be untrusted. 

562 Postconditions: 

563 The packet, story coverage, source citations, generation metadata, and 

564 annotation byte limit agree. The report is not modified. 

565 

566 Essential behavior: 

567 Checks source and story without replaying production proofs or sessions. 

568 """ 

569 if ( 

570 type(report) is not dict 

571 or set(report) != {"schema", "packet", "story", "generation"} 

572 or report["schema"] != REPORT_VERSION 

573 ): 

574 msg = "Unsupported walkthrough report" 

575 raise ValueError(msg) 

576 packet = report["packet"] 

577 validate_packet(packet) 

578 generation = report["generation"] 

579 metadata = Generation.model_validate(generation) 

580 validate_story(packet, report["story"]) 

581 if generation.get("evidence_digest") != packet_identity(packet): 

582 msg = "Walkthrough evidence binding mismatch" 

583 raise ValueError(msg) 

584 for side in ("base", "head"): 

585 if generation.get(f"{side}_sha") != packet["meta"][f"{side}_sha"]: 

586 msg = "Walkthrough generation revision mismatch" 

587 raise ValueError(msg) 

588 for field in ("provider", "model", "destination"): 

589 if ( 

590 not isinstance(generation.get(field), str) 

591 or not generation[field].strip() 

592 ): 

593 msg = "Walkthrough provider metadata is missing" 

594 raise ValueError(msg) 

595 if metadata.usage.calls > MAX_REQUESTS: 

596 msg = "Walkthrough usage exceeds the run request ceiling" 

597 raise ValueError(msg) 

598 if ( 

599 len( 

600 canonical_bytes( 

601 {"story": report["story"], "generation": generation} 

602 ) 

603 ) 

604 > MAX_ANNOTATION_BYTES 

605 ): 

606 msg = "Walkthrough annotations exceed the 1 MiB limit" 

607 raise ValueError(msg)

640def apply_walkthrough_annotations(report: dict, annotations: dict) -> dict: 

641 """Apply exact-evidence annotations to compiler source after boundary checks. 

642 

643 Args: 

644 report: Compiler report for the annotations' pinned source.

651 them through the result also changes the input annotations. 

652 

653 Preconditions: 

654 ``report`` is compiler output for the exact source pinned by the 

655 annotations. 

656 Postconditions: 

657 The result passes walkthrough validation and retains the source packet. 

658 Essential behavior: 

659 The function reuses the annotation objects; it does not deep-copy them. 

660 

661 Raises: 

662 ValueError: If source identity, revisions, schema, or story differ. 

663 """ 

664 if ( 

665 set(annotations) != {"schema", "story", "generation"} 

666 or annotations["schema"] != "diffstory.annotations.walkthrough.v1" 

667 ): 

668 msg = "Unsupported walkthrough annotations" 

669 raise ValueError(msg) 

670 result = { 

671 "schema": REPORT_VERSION, 

672 "packet": build_packet(report), 

673 "story": annotations["story"], 

674 "generation": annotations["generation"], 

675 } 

676 validate_walkthrough(result) 

677 return result

680class WalkthroughNarrator: 

681 """Write one complete story or bounded parallel parts plus one editor. 

682 

683 Attributes: 

684 provider: Prepared structured-output adapter with explicit capacities. 

685 limits: Run-wide request, concurrency, deadline, and annotation ceilings. 

686 state_dir: Durable budget directory, or None for an ephemeral run. 

687 cache_dir: Request-keyed result cache, or None to disable payload storage. 

688 usage: Measured cumulative usage, including resumed and failed attempts. 

689 timings: Local preparation/assembly times and aggregate provider and 

690 validation call durations. Concurrent call durations can overlap. 

691 cache_hits: Validated cached responses reused during this invocation. 

692 """ 

693 

694 def __init__( # noqa: PLR0913 # These are independent user-visible run controls. 

695 self, 

696 provider: NarrativeProvider, 

697 *, 

698 deadline_seconds: float = DEFAULT_DEADLINE, 

699 concurrency: int = 4, 

700 max_requests: int = 12, 

701 state_dir: Path | None = None, 

702 cache_dir: Path | None = None, 

703 ) -> None: 

704 """Configure bounded generation and its durable run controls. 

705 

706 Args: 

707 provider: Structured-output adapter with explicit model capacities. 

708 deadline_seconds: Finite number greater than zero and at most 240. 

709 Booleans are rejected. 

710 concurrency: Integer from 1 through 4. Booleans are rejected. 

711 max_requests: Integer from 1 through 12. Booleans are rejected. 

712 state_dir: Durable run-state directory, or ``None`` for an 

713 invocation-local run. 

714 cache_dir: Directory for request-keyed result payloads, or ``None`` 

715 to disable payload caching. 

716 

717 Raises: 

718 RunLimitError: If a numeric limit is outside its accepted range. 

719 

720 Postconditions: 

721 Durable resumption retains spent attempts and the original deadline. 

722 Essential behavior: 

723 The run ceilings apply to preparation, writers, retries, and editing. 

724 """ 

725 self.provider = provider 

726 self.limits = RunLimits( 

727 requests=max_requests, 

728 concurrency=concurrency, 

729 seconds=deadline_seconds, 

730 ) 

731 self.state_dir, self.cache_dir = state_dir, cache_dir 

732 self.usage: RunUsage | WalkthroughRuntime = RunUsage() 

733 self.timings: dict[str, float] = {} 

734 self.cache_hits = 0 

735 self._prepared: WalkthroughPlan | None = None 

736 

737 def prepare(self, report: dict) -> WalkthroughPlan: 

738 """Build the complete bounded plan before any provider receives source. 

739 

740 Args: 

741 report: Compiler report with exact pinned source evidence. 

742 

743 Returns: 

744 Capacity-checked writing requests and the maximum editor envelope. 

745 

746 Raises: 

747 ValueError: If coherent source cannot fit the request/call ceilings. 

748 TimeoutError: If local preparation exhausts the narration deadline. 

749 

750 Postconditions: 

751 This plan replaces any earlier prepared plan. Usage and cache-hit 

752 counters are reset. If preparation fails, no plan remains prepared. 

753 Essential behavior: 

754 Only the most recently prepared plan can be executed. 

755 """ 

756 started = time.monotonic() 

757 from .walkthrough_plan import prepare_plan 

758 

759 self._prepared = None 

760 self.cache_hits = 0 

761 self.usage = RunUsage() 

762 packet = build_packet(report) 

763 if not packet["changes"] and not packet["regions"]: 

764 msg = "No supplied changes to narrate" 

765 raise WalkthroughError(msg) 

766 plan = prepare_plan(packet, self.provider) 

767 calls = len(plan.writers) + int(not plan.single) 

768 if calls > self.limits.requests: 

769 msg = "Walkthrough plan exceeds the total request limit before source transfer" 

770 raise WalkthroughError(msg) 

771 self.timings = {"preparation_seconds": time.monotonic() - started} 

772 if self.timings["preparation_seconds"] >= self.limits.seconds: 

773 msg = "Narration deadline exceeded during local preparation" 

774 raise RunDeadlineError(msg) 

775 self._prepared = plan 

776 return plan 

777 

778 def generate(self, report: dict) -> dict: 

779 """Prepare and generate a complete flat report without mutating input.""" 

780 self.prepare(report) 

781 return self.generate_prepared() 

782 

783 def generate_prepared(self) -> dict: 

784 """Execute the approved plan, sharing budgets across writers and editor. 

785 

786 Returns: 

787 A validated flat report. Confirmation wait is excluded; preparation, 

788 provider work, validation, cache handling, and assembly are bounded. 

789 

790 Raises: 

791 ValueError: If evidence/configuration changed, limits fail, or model 

792 results fail structural validation after bounded retries. 

793 TimeoutError: If the original run deadline is exceeded. 

794 

795 Preconditions: 

796 A plan was returned by the most recent successful ``prepare`` call. 

797 Postconditions: 

798 The prepared plan is consumed before execution starts. It stays 

799 consumed if validation, provider work, or assembly later fails. 

800 Essential behavior: 

801 Each prepared plan is one-use. Call ``prepare`` again before a retry; 

802 durable state still retains the earlier attempts and deadline. 

803 The assembled report receives one complete validation pass before 

804 it is returned, so its immediate caller can reuse that result 

805 locally without persisting a trust flag. 

806 """ 

807 started = time.monotonic() - self.timings.get("preparation_seconds", 0) 

808 from .walkthrough_plan import EDITOR_SYSTEM 

809 from .walkthrough_plan import Editorial 

810 from .walkthrough_plan import assemble_story 

811 from .walkthrough_plan import check_capacity 

812 from .walkthrough_plan import editorial_input 

813 from .walkthrough_plan import validate_part 

814 

815 if self._prepared is None: 

816 msg = "Prepare a walkthrough before generation" 

817 raise WalkthroughError(msg) 

818 plan = copy.deepcopy(self._prepared) 

819 self._prepared = None 

820 packet = plan.packet 

821 if ( 

822 packet_identity(packet) != plan.evidence_digest 

823 or provider_identity(self.provider) != plan.provider 

824 ): 

825 msg = "Prepared walkthrough evidence or provider configuration changed" 

826 raise WalkthroughError(msg) 

827 run_identity = digest( 

828 { 

829 "evidence": plan.evidence_digest, 

830 "provider": plan.provider, 

831 "writers": [ 

832 request_key(writer.body, plan.provider) 

833 for writer in plan.writers 

834 ], 

835 "editor_system": EDITOR_SYSTEM, 

836 "editor_schema": Editorial.model_json_schema(), 

837 "limits": asdict(self.limits), 

838 } 

839 ) 

840 self.timings["preparation_seconds"] = time.monotonic() - started 

841 runtime = WalkthroughRuntime( 

842 self.provider, 

843 run_identity, 

844 limits=self.limits, 

845 state_dir=self.state_dir, 

846 cache_dir=self.cache_dir, 

847 preparation_seconds=self.timings["preparation_seconds"], 

848 ) 

849 self.usage = runtime 

850 with runtime: 

851 if plan.single: 

852 story = runtime.call( 

853 plan.writers[0].body, partial(validate_story, packet) 

854 ) 

855 else: 

856 parts = runtime.parallel( 

857 [ 

858 ( 

859 writer.body, 

860 partial( 

861 validate_part, 

862 writer.evidence, 

863 packet=packet, 

864 writer_index=index, 

865 provider=self.provider, 

866 ), 

867 ) 

868 for index, writer in enumerate(plan.writers) 

869 ] 

870 ) 

871 edit_started = time.monotonic() 

872 data, passages = editorial_input(packet, parts, self.provider) 

873 body = self.provider.prepare( 

874 EDITOR_SYSTEM, 

875 data, 

876 "walkthrough_editor", 

877 Editorial.model_json_schema(), 

878 OUTPUT_TOKENS, 

879 ) 

880 check_capacity(self.provider, body) 

881 if len(body) > plan.editorial_bound_bytes: 

882 msg = "Editorial request exceeds its preflighted envelope" 

883 raise WalkthroughError(msg) 

884 self.timings["preparation_seconds"] += ( 

885 time.monotonic() - edit_started 

886 ) 

887 story = runtime.call( 

888 body, partial(assemble_story, packet, passages) 

889 ) 

890 assembly_started = time.monotonic() 

891 if provider_identity(self.provider) != plan.provider: 

892 msg = "Provider configuration changed during narration" 

893 raise WalkthroughError(msg) 

894 result = { 

895 "schema": REPORT_VERSION, 

896 "packet": packet, 

897 "story": story, 

898 "generation": { 

899 "version": GENERATION_VERSION, 

900 "state": "complete", 

901 "provider": self.provider.name, 

902 "model": self.provider.model, 

903 "destination": self.provider.destination, 

904 "base_sha": packet["meta"]["base_sha"], 

905 "head_sha": packet["meta"]["head_sha"], 

906 "evidence_digest": plan.evidence_digest, 

907 "usage": runtime.report(), 

908 "writing_requests": len(plan.writers), 

909 "editorial_requests": int(not plan.single), 

910 "cache_hits": runtime.cache_hits, 

911 "new_requests": runtime.new_calls, 

912 }, 

913 } 

914 validate_walkthrough(result) 

915 runtime.finish() 

916 runtime.remaining() 

917 self.timings.update(runtime.record["timings"]) 

918 self.timings["validation_seconds"] += ( 

919 time.monotonic() - assembly_started 

920 ) 

921 runtime.remaining() 

922 self.cache_hits = runtime.cache_hits 

923 self.timings["narration_seconds"] = time.monotonic() - started 

924 return result
